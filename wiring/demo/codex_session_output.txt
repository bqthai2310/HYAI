from __future__ import annotations

import math
from collections import deque
from typing import Deque, Dict


class RollingLatencyTracker:
    def __init__(self, window_size: int = 1000):
        if window_size <= 0:
            raise ValueError("window_size must be a positive integer")
        self._window_size = window_size
        self._samples: Deque[float] = deque(maxlen=window_size)

    def record(self, latency_ms: float) -> None:
        if latency_ms < 0:
            raise ValueError("latency_ms must be non-negative")
        self._samples.append(float(latency_ms))

    @property
    def count(self) -> int:
        return len(self._samples)

    def clear(self) -> None:
        self._samples.clear()

    def get_stats(self) -> Dict[str, float]:
        if self.count == 0:
            raise ValueError("Cannot compute statistics on empty buffer")
        ordered = sorted(self._samples)
        return {
            "min": ordered[0],
            "max": ordered[-1],
            "mean": math.fsum(ordered) / len(ordered),
            "p50": _percentile(ordered, 50),
            "p95": _percentile(ordered, 95),
            "p99": _percentile(ordered, 99),
        }

+"""Rolling latency statistics over a fixed-size window of samples.
+
+The tracker keeps the most recent ``window_size`` latency observations and
+derives summary statistics (including percentiles) from that window only.
+"""
+
+from __future__ import annotations
+
+import math
+from collections import deque
+from typing import Deque, Dict, Sequence
+
+
+def _percentile(ordered: Sequence[float], pct: float) -> float:
+    """Return the ``pct`` percentile of an ascending-sorted sequence.
+
+    Uses linear interpolation between the two nearest ranks, which matches
+    the inclusive method used by ``numpy.percentile`` by default. A single
+    sample is its own percentile for every requested percentage.
+    """
+    n = len(ordered)
+    if n == 1:
+        return ordered[0]
+
+    rank = (n - 1) * (pct / 100.0)
+    lower = math.floor(rank)
+    upper = math.ceil(rank)
+    if lower == upper:
+        return ordered[int(rank)]
+    return ordered[lower] + (ordered[upper] - ordered[lower]) * (rank - lower)
+
+
+class RollingLatencyTracker:
+    """Collect latency samples in a bounded FIFO window.
+
+    Args:
+        window_size: Maximum number of samples retained. Older samples are
+            evicted first-in-first-out once the window is full.
+
+    Raises:
+        ValueError: If ``window_size`` is not positive.
+    """
+
+    def __init__(self, window_size: int = 1000):
+        if window_size <= 0:
+            raise ValueError("window_size must be a positive integer")
+        self._window_size = window_size
+        self._samples: Deque[float] = deque(maxlen=window_size)
+
+    @property
+    def window_size(self) -> int:
+        return self._window_size
+
+    def record(self, latency_ms: float) -> None:
+        """Add one latency observation to the window.
+
+        Raises:
+            ValueError: If ``latency_ms`` is negative.
+        """
+        if latency_ms < 0:
+            raise ValueError("latency_ms must be non-negative")
+        self._samples.append(float(latency_ms))
+
+    @property
+    def count(self) -> int:
+        """Number of samples currently held in the window."""
+        return len(self._samples)
+
+    def clear(self) -> None:
+        """Discard all buffered samples."""
+        self._samples.clear()
+
+    def get_stats(self) -> Dict[str, float]:
+        """Compute summary statistics over the current window.
+
+        Returns:
+            A dict with keys ``min``, ``max``, ``mean``, ``p50``, ``p95``
+            and ``p99``.

+"""Rolling latency statistics over a fixed-size window of samples.
+
+The tracker keeps the most recent ``window_size`` latency observations and
+derives summary statistics (including percentiles) from that window only.
+"""
+
+from __future__ import annotations
+
+import math
+from collections import deque
+from typing import Deque, Dict, Sequence
+
+
+def _percentile(ordered: Sequence[float], pct: float) -> float:
+    """Return the ``pct`` percentile of an ascending-sorted sequence.
+
+    Uses linear interpolation between the two nearest ranks, which matches
+    the inclusive method used by ``numpy.percentile`` by default. A single
+    sample is its own percentile for every requested percentage.
+    """
+    n = len(ordered)
+    if n == 1:
+        return ordered[0]
+
+    rank = (n - 1) * (pct / 100.0)
+    lower = math.floor(rank)
+    upper = math.ceil(rank)
+    if lower == upper:
+        return ordered[int(rank)]
+    return ordered[lower] + (ordered[upper] - ordered[lower]) * (rank - lower)
+
+
+class RollingLatencyTracker:
+    """Collect latency samples in a bounded FIFO window.
+
+    Args:
+        window_size: Maximum number of samples retained. Older samples are
+            evicted first-in-first-out once the window is full.
+
+    Raises:
+        ValueError: If ``window_size`` is not positive.
+    """
+
+    def __init__(self, window_size: int = 1000):
+        if window_size <= 0:
+            raise ValueError("window_size must be a positive integer")
+        self._window_size = window_size
+        self._samples: Deque[float] = deque(maxlen=window_size)
+
+    @property
+    def window_size(self) -> int:
+        return self._window_size
+
+    def record(self, latency_ms: float) -> None:
+        """Add one latency observation to the window.
+
+        Raises:
+            ValueError: If ``latency_ms`` is negative.
+        """
+        if latency_ms < 0:
+            raise ValueError("latency_ms must be non-negative")
+        self._samples.append(float(latency_ms))
+
+    @property
+    def count(self) -> int:
+        """Number of samples currently held in the window."""
+        return len(self._samples)
+
+    def clear(self) -> None:
+        """Discard all buffered samples."""
+        self._samples.clear()
+
+    def get_stats(self) -> Dict[str, float]:
+        """Compute summary statistics over the current window.
+
+        Returns:
+            A dict with keys ``min``, ``max``, ``mean``, ``p50``, ``p95``
+            and ``p99``.
+
+        Raises:
+            ValueError: If the window is empty.
+        """
+        if self.count == 0:
+            raise ValueError("Cannot compute statistics on empty buffer")
+
+        ordered = sorted(self._samples)
+        return {
+            "min": ordered[0],
